import React, { useState } from 'react';
import {
Sparkles, Image, BookOpen, PenTool, FileText,
Settings, Send, Download, Plus, Trash2, Moon, Sun, Palette
} from 'lucide-react';

export default function GPowerAI() {
const [activeTab, setActiveTab] = useState('study');
const [themeColor, setThemeColor] = useState('orange'); // orange, blue, emerald, purple
const [prompt, setPrompt] = useState('');
const [chatHistory, setChatHistory] = useState([
{ sender: 'ai', text: 'హలో బిట్టు! నేను GPower AI. మీకు చదువులో సహాయం కావాలన్నా, ఇమేజెస్ క్రియేట్ చేయాలన్నా లేదా కథలు/ఐడియాలు కావాలన్నా అడగండి!' }
]);
const [notes, setNotes] = useState([
{ id: 1, title: 'Physics Formulas', content: 'Important equations for exams...' }
]);
const [newNoteTitle, setNewNoteTitle] = useState('');
const [newNoteContent, setNewNoteContent] = useState('');

// Dynamic Theme Colors
const themeClasses = {
orange: 'from-orange-500 to-amber-600 border-orange-500 text-orange-500 bg-orange-500',
blue: 'from-blue-500 to-indigo-600 border-blue-500 text-blue-500 bg-blue-500',
emerald: 'from-emerald-500 to-teal-600 border-emerald-500 text-emerald-500 bg-emerald-500',
purple: 'from-purple-500 to-pink-600 border-purple-500 text-purple-500 bg-purple-500'
};

const handleSendMessage = () => {
if (!prompt.trim()) return;
setChatHistory(prev => [...prev, { sender: 'user', text: prompt }]);

// Simulated AI response
setTimeout(() => {
setChatHistory(prev => [...prev, {
sender: 'ai',
text: `మీరు అడిగిన "${prompt}" కి సంబంధించి పూర్తి వివరణ ఇక్కడ పొందుపరచబడింది. మీ సందేహాలను ఇంగ్లీష్ లేదా తెలుగులో ఇంకా వివరంగా అడగవచ్చు.`
}]);
}, 1000);

setPrompt('');
};

const addNote = () => {
if (!newNoteTitle.trim()) return;
setNotes([...notes, { id: Date.now(), title: newNoteTitle, content: newNoteContent }]);
setNewNoteTitle('');
setNewNoteContent('');
};

const deleteNote = (id) => {
setNotes(notes.filter(note => note.id !== id));
};

return (
<div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col font-sans">

{/* Header / Navbar */}
<header className="border-b border-slate-800 bg-slate-900/80 backdrop-blur sticky top-0 z-50 px-6 py-4 flex justify-between items-center">
<div className="flex items-center gap-3">
<div className={`p-2 rounded-xl bg-gradient-to-r ${themeClasses[themeColor]} text-white shadow-lg`}>
<Sparkles className="w-6 h-6" />
</div>
<div>
<h1 className="text-xl font-bold tracking-wide">GPower AI</h1>
<p className="text-xs text-slate-400">Study & Creative AI Companion</p>
</div>
</div>

{/* Theme Selector */}
<div className="flex items-center gap-2 bg-slate-800 p-1.5 rounded-lg border border-slate-700">
<Palette className="w-4 h-4 text-slate-400 ml-1" />
{['orange', 'blue', 'emerald', 'purple'].map(color => (
<button
key={color}
onClick={() => setThemeColor(color)}
className={`w-5 h-5 rounded-full capitalize transition-transform ${
color === 'orange' ? 'bg-orange-500' :
color === 'blue' ? 'bg-blue-500' :
color === 'emerald' ? 'bg-emerald-500' : 'bg-purple-500'
} ${themeColor === color ? 'scale-125 ring-2 ring-white' : 'opacity-60 hover:opacity-100'}`}
/>
))}
</div>
</header>

{/* Main Container */}
<div className="flex-1 flex flex-col md:flex-row max-w-7xl w-full mx-auto p-4 gap-4">

{/* Sidebar Navigation */}
<nav className="w-full md:w-64 bg-slate-900 border border-slate-800 rounded-2xl p-4 flex md:flex-col gap-2">
<button
onClick={() => setActiveTab('study')}
className={`flex items-center gap-3 w-full p-3 rounded-xl font-medium transition ${activeTab === 'study' ? 'bg-slate-800 text-white border-l-4 border-orange-500' : 'text-slate-400 hover:bg-slate-800/50'}`}>
<BookOpen className="w-5 h-5" /> Study Assistant
</button>
<button
onClick={() => setActiveTab('image')}
className={`flex items-center gap-3 w-full p-3 rounded-xl font-medium transition ${activeTab === 'image' ? 'bg-slate-800 text-white border-l-4 border-orange-500' : 'text-slate-400 hover:bg-slate-800/50'}`}>
<Image className="w-5 h-5" /> Image Generator
</button>
<button
onClick={() => setActiveTab('story')}
className={`flex items-center gap-3 w-full p-3 rounded-xl font-medium transition ${activeTab === 'story' ? 'bg-slate-800 text-white border-l-4 border-orange-500' : 'text-slate-400 hover:bg-slate-800/50'}`}>
<PenTool className="w-5 h-5" /> Story & Ideas
</button>
<button
onClick={() => setActiveTab('notes')}
className={`flex items-center gap-3 w-full p-3 rounded-xl font-medium transition ${activeTab === 'notes' ? 'bg-slate-800 text-white border-l-4 border-orange-500' : 'text-slate-400 hover:bg-slate-800/50'}`}>
<FileText className="w-5 h-5" /> Study Notes
</button>
</nav>

{/* Content Area */}
<main className="flex-1 bg-slate-900 border border-slate-800 rounded-2xl p-6 flex flex-col justify-between">

{/* TAB 1: Study Assistant */}
{activeTab === 'study' && (
<div className="flex flex-col h-full justify-between">
<div>
<h2 className="text-lg font-semibold mb-2 flex items-center gap-2">
<BookOpen className="w-5 h-5 text-orange-500" /> Study & Concept Explainer
</h2>
<p className="text-sm text-slate-400 mb-4">మీ చదువుకు సంబంధించి ఎలాంటి పాఠాలైనా, కాన్సెప్ట్‌లైనా ఇక్కడ అడగండి.</p>

<div className="space-y-3 max-h-[400px] overflow-y-auto pr-2">
{chatHistory.map((msg, idx) => (
<div key={idx} className={`p-3 rounded-xl max-w-[80%] ${msg.sender === 'user' ? 'bg-orange-600 text-white ml-auto' : 'bg-slate-800 text-slate-200'}`}>
{msg.text}
</div>
))}
</div>
</div>

<div className="flex gap-2 mt-4">
<input
type="text"
value={prompt}
onChange={(e) => setPrompt(e.target.value)}
onKeyDown={(e) => e.key === 'Enter' && handleSendMessage()}
placeholder="ఏదైనా టాపిక్ లేదా ప్రశ్న టైప్ చేయండి (English / తెలుగు)..."
className="flex-1 bg-slate-950 border border-slate-700 rounded-xl px-4 py-3 text-sm focus:outline-none focus:border-orange-500"
/>
<button onClick={handleSendMessage} className="bg-orange-500 hover:bg-orange-600 text-white px-5 rounded-xl font-medium transition flex items-center justify-center">
<Send className="w-5 h-5" />
</button>
</div>
</div>
)}

{/* TAB 2: Image Generator */}
{activeTab === 'image' && (
<div>
<h2 className="text-lg font-semibold mb-2 flex items-center gap-2">
<Image className="w-5 h-5 text-orange-500" /> AI Image Generator
</h2>
<p className="text-sm text-slate-400 mb-4">మీ ఊహకు తగ్గట్టుగా HD ఇమేజెస్ డిజైన్ చేసుకోండి.</p>

<div className="flex gap-2 mb-4">
<input
type="text"
placeholder="ఉదాహరణ: A futuristic library with glowing books in orange neon light..."
className="flex-1 bg-slate-950 border border-slate-700 rounded-xl px-4 py-3 text-sm focus:outline-none focus:border-orange-500"
/>
<button className="bg-orange-500 hover:bg-orange-600 text-white px-6 rounded-xl font-medium transition flex items-center gap-2">
<Sparkles className="w-4 h-4" /> Generate
</button>
</div>

<div className="border-2 border-dashed border-slate-800 rounded-2xl h-64 flex flex-col items-center justify-center text-slate-500">
<Image className="w-12 h-12 mb-2 opacity-50" />
<p>ఇక్కడ మీ AI ఇమేజ్ కనిపిస్తుంది</p>
</div>
</div>
)}

{/* TAB 3: Story & Ideas */}
{activeTab === 'story' && (
<div>
<h2 className="text-lg font-semibold mb-2 flex items-center gap-2">
<PenTool className="w-5 h-5 text-orange-500" /> Story & Idea Generator
</h2>
<p className="text-sm text-slate-400 mb-4">కొత్త కథలు, ప్రాజెక్ట్ ఐడియాలు లేదా క్రియేటివ్ స్క్రిప్ట్స్ కోసం ఉపయోగించండి.</p>
<textarea
rows="4"
placeholder="మీ కథాంశం లేదా ఐడియా గురించి ఇక్కడ రాయండి..."
className="w-full bg-slate-950 border border-slate-700 rounded-xl p-4 text-sm focus:outline-none focus:border-orange-500 mb-4"
></textarea>
<button className="bg-orange-500 hover:bg-orange-600 text-white px-6 py-2.5 rounded-xl font-medium transition">
Generate Ideas
</button>
</div>
)}

{/* TAB 4: Notes Vault */}
{activeTab === 'notes' && (
<div>
<h2 className="text-lg font-semibold mb-2 flex items-center gap-2">
<FileText className="w-5 h-5 text-orange-500" /> Study Notes Vault
</h2>
<p className="text-sm text-slate-400 mb-4">మీ ఇంపార్టెంట్ పాయింట్లను ఇక్కడ సేవ్ చేసుకోండి.</p>

<div className="flex flex-col gap-2 mb-6">
<input
type="text"
placeholder="Note Title..."
value={newNoteTitle}
onChange={(e) => setNewNoteTitle(e.target.value)}
className="bg-slate-950 border border-slate-700 rounded-xl px-4 py-2 text-sm focus:outline-none focus:border-orange-500"
/>
<textarea
placeholder="Note details..."
value={newNoteContent}
onChange={(e) => setNewNoteContent(e.target.value)}
className="bg-slate-950 border border-slate-700 rounded-xl p-4 text-sm focus:outline-none focus:border-orange-500"
></textarea>
<button onClick={addNote} className="bg-orange-500 hover:bg-orange-600 text-white px-4 py-2 rounded-xl font-medium transition flex items-center gap-2 w-fit">
<Plus className="w-4 h-4" /> Save Note
</button>
</div>

<div className="grid grid-cols-1 md:grid-cols-2 gap-3">
{notes.map(note => (
<div key={note.id} className="bg-slate-950 border border-slate-800 p-4 rounded-xl flex justify-between items-start">
<div>
<h4 className="font-semibold text-white">{note.title}</h4>
<p className="text-xs text-slate-400 mt-1">{note.content}</p>
</div>
<button onClick={() => deleteNote(note.id)} className="text-slate-500 hover:text-red-400">
<Trash2 className="w-4 h-4" />
</button>
</div>
))}
</div>
</div>
)}

</main>
</div>
</div>
);
}
